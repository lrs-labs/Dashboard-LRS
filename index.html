// Dashboard Proyectos Técnicos — endpoint de datos (v3)
const HOJA = 'REGISTRO_CASOS', TAB = 'TABULADOR';

function doGet(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(HOJA);
  const last = sh.getLastRow();
  const vals = sh.getRange(3, 1, last - 2, 47).getValues();   // A..AU
  const tz = Session.getScriptTimeZone();
  const txt = v => String(v == null ? '' : v).trim();
  const num = v => typeof v === 'number' ? v : (parseFloat(String(v).replace(/[^0-9.\-]/g, '')) || 0);
  const day = v => (v instanceof Date && !isNaN(v)) ? Utilities.formatDate(v, tz, 'yyyy-MM-dd') : '';
  const rows = [];
  vals.forEach(r => {
    if (!(r[1] instanceof Date) || isNaN(r[1])) return;
    rows.push([ day(r[1]), txt(r[5]), txt(r[6]), txt(r[7]), txt(r[8]), txt(r[9]), txt(r[10]), txt(r[11]), txt(r[12]),
      txt(r[13]), txt(r[14]), txt(r[15]), txt(r[16]), txt(r[17]), day(r[18]), txt(r[20]), txt(r[21]), txt(r[22]),
      num(r[23]), num(r[24]), num(r[25]), num(r[26]), num(r[30]), txt(r[34]), day(r[35]), txt(r[36]), txt(r[37]),
      txt(r[38]), txt(r[39]), txt(r[46]) ]);      // ...AI, AJ, AK, AL, AM, AN, AU (QUALITY)
  });
  // TABULADOR como lista de celdas no vacías [fila, col, valor]
  const tab = [];
  const ts = ss.getSheetByName(TAB);
  if (ts) ts.getDataRange().getDisplayValues().forEach((r, ri) => r.forEach((v, ci) => { if (String(v).trim() !== '') tab.push([ri, ci, String(v).trim()]); }));
  const body = JSON.stringify({ generatedAt: new Date().toISOString(), schema: 3, total: rows.length, rows, tab });
  const cb = e && e.parameter && e.parameter.callback;
  if (cb) return ContentService.createTextOutput(cb + '(' + body + ')').setMimeType(ContentService.MimeType.JAVASCRIPT);
  return ContentService.createTextOutput(body).setMimeType(ContentService.MimeType.JSON);
}

Notas de lectura

  • Solo se cuentan registros con STATUS = COMPLETED.
  • El bloque Project Status es la excepción: cuenta todos los registros del período (COMPLETED, CANCELED, IN PROGRESS, PENDING) cruzados con CATEGORY, con los mismos filtros.
  • SUBTOTAL PROJECT = $TP + $TA1 + $TA2 + $PRV. "Pago a técnicos" excluye el $PRV. El sheet tiene hasta TA2; si se agrega una columna TA3 hay que mapearla.
  • Documentación: 360 aplica a MITIGATION y MITIGATION PICKUP; hoja de inspección solo a MITIGATION; hoja de trabajo y temperaturas a MITIGATION y REMEDIATION; email report a todas. N/A = no aplica y no entra en el porcentaje.
  • INSURANCE, ¿LIMITE?, REP y PRV se cargan sobre todo en los casos NEW; "-" se muestra como "sin dato". "ALL STATE"/"ALLSTATE" y "STATE FARM"/"STATEFARM" se unifican.
  • "Vencido" en retiros se calcula contra la fecha de hoy en tu computadora. MODALIDAD, HORAS, AF/AG/AH y AO–AT/AV quedan fuera.
